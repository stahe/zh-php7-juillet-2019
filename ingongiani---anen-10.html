    上一版演示表明，所有应用用户共用的税务数据应存储在 [Application] 作用域内存中。我们将使用 Redis 服务器 [https://redis.io] 来实现这一功能。

    20.2. Redis的安装

    Laragon 默认未启用 Redis 服务器。因此，首先需要将其启用：

    • 在 [3] 中，启用服务器 [Redis]；
    • 在 [4] 中，保留 [6379] 端口，这是 Redis 客户端默认使用的端口；

    启用 Redis 后，Laragon 服务将自动重启：

    20.3. 命令行模式下的 Redis 客户端

    可以通过命令模式查询 Redis 服务器。打开一个 Laragon 终端（参见链接部分）：

    • 在 [1] 中，命令 [redis-cli] 会启动 Redis 服务器的命令行客户端；

    截至2019年7月，Redis客户端可通过172条命令与服务器进行交互。 其中一个命令 [command count] [2]，会显示这个数字 [3]。

    我们仅介绍在PHP应用程序中需要用到的命令。我们将Redis用于唯一一个目的：在Redis内存中存储一个数组[‘attribut’=>’valeur’]。 这可以通过 Redis 命令 [set attribut valeur] [4] 实现。随后可通过命令 [get attribut] [5] 检索该值。这就是我们所需的一切。

    可能需要清空Redis的内存。这可以通过命令 [flushdb] [6] 来实现。 随后若查询属性 [titre] [7] 的值，将返回引用 [nil] [8]，表明未找到该属性。 也可以使用命令 [exists] [9-10] 来验证属性的存在性。

    要退出 Redis 客户端，请输入命令 [quit] [11]。

// 严格遵守函数参数的声明类型
declare (strict_types=1);

// 命名空间
namespace Application;

// PHP 的错误处理
ini_set("display_errors", "0");

// 配置文件路径
define("CONFIG_FILENAME", "Data/config-server.json");
// 类别名
use \Application\ServerDaoWithSession as ServerDaoWithRedis;

// 会话
$session = new Session();
$session->start();
…
…
// 第一个日志
$logger->write("\n---nouvelle requête\n");

// 获取当前请求
$request = Request::createFromGlobals();
// 仅首次认证
if (!$session->has("user")) {
  …
} else {
  // 日志
  $logger->write("Authentification prise en session…\n");
}

// 用户有效 - 验证接收到的参数
$erreurs = [];
// 必须有三个参数 GET
$method = strtolower($request->getMethod());
…

// 有错误吗？
if ($erreurs) {
// 向客户端发送 400 错误代码 HTTP_BAD_REQUEST
  sendResponse($response, ["erreurs" => $erreurs], Response::HTTP_BAD_REQUEST, [], $logger);
  // 完成
  exit;
} else {
  // 日志
  $logger->write("paramètres ['marié'=>$marié, 'enfants'=>$enfants, 'salaire'=>$salaire] valides\n");
}

// 已具备所有必要条件
// Redis
\Predis\Autoloader::register();
try {
  // 客户端 [predis]
  $redis = new \Predis\Client();
  // 连接服务器以检查其是否在线
  $redis->connect();
} catch (\Predis\Connection\ConnectionException $ex) {
  // 内部服务器错误
  doInternalServerError("[redis], " . utf8_encode($ex->getMessage()), $response, $config['adminMail'], $logger);
  // 完成
  exit;
}

// 创建图层 [dao]
if (!$redis->get("taxAdminData")) {
  // 从数据库中提取税务数据
  $logger->write("données fiscales prises en base de données\n");
  try {
    // 构建图层 [dao]
    $dao = new ServerDaoWithRedis($config["databaseFilename"], NULL);
    // 将税务数据放入作用域内存 [application]
    // 将隐式调用方法 [TaxAdminData]->__toString
    $redis->set("taxAdminData", $dao->getTaxAdminData());
  } catch (\RuntimeException $ex) {
    // 记录错误
    doInternalServerError("[dao], " . utf8_encode($ex->getMessage()), $response, $config['adminMail'], $logger, $redis);
    // 完成
    exit;
  }
} else {
  // 从作用域内存中获取税务数据 [application]
  $arrayOfAttributes = \json_decode($redis->get("taxAdminData"), true);
  $taxAdminData = (new TaxAdminData())->setFromArrayOfAttributes($arrayOfAttributes);
  // 实例化层 [dao]
  $dao = new ServerDaoWithRedis(NULL, $taxAdminData);
  // 日志
  $logger->write("données fiscales prises dans redis\n");
}
// 创建层 [métier]
$métier = new ServerMetier($dao);
// 计算税款
$result = $métier->calculerImpot($marié, (int) $enfants, (int) $salaire);
// 返回结果
sendResponse($response, $result, Response::HTTP_OK, [], $logger, $redis);
// 结束
exit;

function doInternalServerError(string $message, Response $response, array $infos,
  Logger $logger = NULL, \Predis\Client $predisClient = NULL) {
  // $message：错误信息
  // $response：响应 HTTP
  // $infos：发送邮件的信息表
  // $result：结果表
  // $logger：应用程序日志记录器
  // $predisClient：一个客户端 [predis]
  //
  // 向管理员发送一封邮件
  // SendAdminMail 拦截所有异常并自行记录
  $infos['message'] = $message;
  $sendAdminMail = new SendAdminMail($infos, $logger);
  $sendAdminMail->send();
  // 向客户端发送 500 错误代码
  sendResponse($response, ["erreur" => $message], Response::HTTP_INTERNAL_SERVER_ERROR, [], $logger, $predisClient);
}

// 向客户端发送响应 HTTP 的函数
function sendResponse(Response $response, array $result, int $statusCode,
  array $headers, Logger $logger = NULL, \Predis\Client $predisClient = NULL) {
  // $response：响应 HTTP
  // $result：结果表
  // $statusCode：响应状态 HTTP
  // $headers：应放入响应中的 HTTP 头部
  // $logger：应用程序日志记录器
  // $predisClient：一个客户端 [predis]
  //
  // 状态 HTTTP
  $response->setStatusCode($statusCode);
  // 正文
  $body = \json_encode(["réponse" => $result], JSON_UNESCAPED_UNICODE);
  $response->setContent($body);
  // 头部
  $response->headers->add($headers);
  // 发送
  $response->send();
  // 日志
  if ($logger != NULL) {
    $logger->write("$body\n");
    $logger->close();
  }
  // 关闭连接[redis]
  if ($predisClient != NULL) {
    $predisClient->disconnect();
  }
}

    注释

    • 第 15 行：为类 [\Application\ServerDaoWithSession] 赋予别名 [ServerDaoWithRedis]，以反映服务器脚本实现的变更；
    • 第 18-19 行：会话被保留。这里有两点信息需要记住：
      • 用户已成功认证。该信息属于 [session] 作用域：它与特定用户相关联，对其他用户无效；
      • 税务管理数据。该信息作用域为 [application]：它不与特定用户相关联，但对所有用户均有效；
    • 第54-64行：创建客户端[redis]，该客户端将与服务器[redis]进行通信。 该客户端将通过服务器的默认端口进行通信。如果服务器未使用默认端口，或者未位于 [localhost] 机器上，则需将这些信息传递给 [\Predis\Client] 类的构造函数；
    • 第 59 行：立即将客户端连接到服务器以检测其响应；
    • 第 60-65 行：若连接 Redis 服务器失败，则向客户端发送错误响应，并向应用程序管理员发送一封邮件；
    • 第67行：向[redis]服务器查询键值[taxAdminData]。若未找到，则从数据库中获取税务数据（第72行）；
    • 第75行： 键 [taxAdminData] 被放置在与变量 [$taxAdminData] 中的字符串 jSON 关联的内存 [redis] 中，该变量是一个类型为 QZXW2HTMLP005326Z 的对象QX。方法 [$redis→set] 期望密钥的值为字符串。因此，它将尝试将类型为 [TaxAdminData] 的对象转换为类型 [string]。 此时将隐式调用方法 [TaxAdminData->__toString]。该方法会生成对象 [TaxAdminData] 的字符串 jSON；
    • 第 84 行：键 [taxAdminData] 位于内存中的 [redis] 位置，因此我们获取其值。我们知道这是对象 [TaxAdminData] 的字符串 jSON。 随后对该字符串进行解码，以获取一个属性数组；
    • 第 85 行：基于该数组，实例化了一个新的 [TaxAdminData] 对象；
    • 第87行：实例化[dao]层；
