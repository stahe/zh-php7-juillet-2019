    函数 [check] 如下：

    
private function check($value): \stdClass {
    // $value 是一个元素数组或单个元素
    // 创建数组
    if (!\is_array($value)) {
      $tableau = [$value];
    } else {
      $tableau = $value;
    }
    // 将类型未知的元素数组转换为实数数组
    $newTableau = [];
    $result = new \stdClass();
    // 数组中的元素必须是正数或零
    $modèle = '/^\s*([+]?)\s*(\d+\.\d*|\.\d+|\d+)\s*$/';
    for ($i = 0; $i < count($tableau); $i ++) {
      if (preg_match($modèle, $tableau[$i])) {
        // 将浮点数放入 newTableau
        $newTableau[] = (float) $tableau[$i];
      } else {
        // 记录错误
        $result->erreur = TRUE;
        // 退出
        return $result;
      }
    }
    // 返回结果
    $result->erreur = FALSE;
    if (!\is_array($value)) {
      // 单个值
      $result->value = $newTableau[0];
    } else {
      // 返回值列表
      $result->value = $newTableau;
    }
    return $result;
  }

    每个 setter 都以 [return $this] 操作结尾。这使得能够按以下方式初始化属性：

    $data→setLimites($limites)→setCoeffR($coeffR)→setCoeffN($coeffN) ;

    8.5. 类 [Utilitaires]

    此外，我们在文件 [Utilitaires.php] 中定义了一个类 [Utilitaires]：

    
<?php

// 命名空间
namespace Application;

// 一个实用函数类
abstract class Utilitaires {

  public static function cutNewLinechar(string $ligne): string {
    // 如果存在，则删除 $ligne 的行尾标记
    $longueur = strlen($ligne);  // 行长度
    while (substr($ligne, $longueur - 1, 1) == "\n" or substr($ligne, $longueur - 1, 1) == "\r") {
      $ligne = substr($ligne, 0, $longueur - 1);
      $longueur--;
    }
    // 结束 - 返回该行
    return($ligne);
  }
}

    注释

    • 第 4 行：类 [Utilitaires] 也位于命名空间 [Exemples] 中；
    • 第 9 行：方法 [cutNewLinechar] 会移除作为参数传递的文本中可能存在的换行符。它返回由此形成的新行。需要注意的是，这是一个静态方法，即它将以 [Utilitaires::cutNewLineChar] 的形式被调用；

      $impot2 = $result2["impôt"];
      // 应用家庭商数上限

      if ($enfants < 3) {

      } else {

      }
    } else {

      $result2 = $result1;
    }
    // 取最高税率
    if ($impot1 > $impot2) {
      $impot = $impot1;
      $taux = $result1["taux"];
      $surcôte = $result1["surcôte"];
    } else {
      $surcôte = $impot2 - $impot1 + $result2["surcôte"];
      $impot = $impot2;
      $taux = $result2["taux"];
    }
    // 计算可能的减免额
    $décôte = $this->getDecôte($marié, $salaire, $impot);
    $impot -= $décôte;
    // 计算可能的减税额
    $réduction = $this->getRéduction($marié, $salaire, $enfants, $impot);
    $impot -= $réduction;
    // 结果
    return ["impôt" => floor($impot), "surcôte" => $surcôte, "décôte" => $décôte, "réduction" => $réduction, "taux" => $taux];
  }

    函数 [executeBatchImpots] 如下所示：

    
public function executeBatchImpots(string $usersFileName, string $resultsFileName, string $errorsFileName): void {
    // 一旦处理文件，可能会出现不少错误
    try {
      // 打开文件时出现错误
      $errors = fopen($errorsFileName, "w");
      if (!$errors) {
        throw new ExceptionImpots("Impossible de créer le fichier des erreurs [$errorsFileName]", 10);
      }
      // 打开结果文件
      $results = fopen($resultsFileName, "w");
      if (!$results) {
        throw new ExceptionImpots("Impossible de créer le fichier des résultats [$resultsFileName]", 11);
      }
      // 读取用户数据
      // 每行数据格式为：婚姻状况、子女数、年薪
      $data = fopen($usersFileName, "r");
      if (!$data) {
        throw new ExceptionImpots("Impossible d'ouvrir en lecture les déclarations des contribuables [$usersFileName]", 12);
      }
      // 处理用户数据文件的当前行
      // 该行格式为：婚姻状况、子女数、年薪
      $num = 1;         // 当前行号
      $nbErreurs = 0;   // 遇到的错误数量
      while ($ligne = fgets($data, 100)) {
        // 调试
        //  print "第 " . ($i + 1) . " 行： " . $ligne;
        // 移除可能存在的换行符
        $ligne = Utilitaires::cutNewLineChar($ligne);
        // 提取组成 $ligne 的 3 个字段：已婚:子女:工资
        list($marié, $enfants, $salaire) = explode(",", $ligne);
        // 对其进行验证
        // 婚姻状况必须为“是”或“否”
        $marié = trim(strtolower($marié));
        $erreur = ($marié !== "oui" and $marié !== "non");
        if (!$erreur) {
          // 子女数量必须为整数
          $enfants = trim($enfants);
          if (!preg_match("/^\s*\d+\s*$/", $enfants)) {
            $erreur = TRUE;
          } else {
            $enfants = (int) $enfants;
          }
        }
        if (!$erreur) {
          // 工资为整数，不包含欧元分
          $salaire = trim($salaire);
          if (!preg_match("/^\s*\d+\s*$/", $salaire)) {
            $erreur = TRUE;
          } else {
            $salaire = (int) $salaire;
          }
        }
        // 错误？
        if ($erreur) {
          fputs($errors, "la ligne [$num] du fichier [$usersFileName] est erronée\n");
          $nbErreurs++;
        } else {
          // 计算税额
          $result = $this->calculerImpot($marié, (int) $enfants, (int) $salaire);
          // 将结果写入结果文件
          $result = ["marié" => $marié, "enfants" => $enfants, "salaire" => $salaire] + $result;
          fputs($results, \json_encode($result, JSON_UNESCAPED_UNICODE) . "\n");
        }
        // 下一行
        $num++;
      }
      // 有错误吗？
      if ($nbErreurs > 0) {
        throw new ExceptionImpots("Il y a eu des erreurs", 15);
      }
    } catch (ExceptionImpots $ex) {
      // 重新抛出异常
      throw $ex;
    } finally {
      // 关闭所有文件
      fclose($data);
      fclose($results);
      fclose($errors);
    }
  }

    8.7. [ImpotsWithTaxAdminDataInJsonFile] 类

    抽象类 [AbstractBaseImpots] 未实现接口 [InterfaceImpots] 中的方法 [getTaxAdminData]。因此，我们需要在派生类中定义该方法。 我们在以下派生类 [ImpotsWithTaxAdminDataInJsonFile] 中进行定义：

    
<?php

// 命名空间
namespace Application;

// 类定义 ImpotsWithDataInArrays
class ImpotsWithTaxAdminDataInJsonFile extends AbstractBaseImpots {
  // 一个 Data 类型的属性
  private $taxAdminData;

  // 构造函数
  public function __construct(string $jsonFileName) {
    // 根据文件 jSON 初始化 $this->taxAdminData
    $this->taxAdminData = (new TaxAdminData())->setFromJsonFile($jsonFileName);
  }

  // 返回用于计算税款的数据
  public function getTaxAdminData(): TaxAdminData {
    // 返回属性 [$this->taxAdminData]
    return $this->taxAdminData;
  }

}

    注释

    • 第7行：类[ImpotsWithTaxAdminDataInJsonFile]继承自抽象类[AbstractBaseImpots]。它需要定义其父类未定义的方法[getTaxAdminData]；
    • 第 9 行：属性 [$taxAdminData] 将包含税务部门的数据；
    • 第12-15行：构造函数接收的唯一参数是包含税务数据的文件名jSON；
